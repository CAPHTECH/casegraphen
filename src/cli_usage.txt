usage:
  casegraphen version
  casegraphen --version
  casegraphen graph lint --input <execution.topology.json> --format json|text [--output <path>]
  casegraphen schema list --format json [--output <path>]  (every schema and example this crate embeds, tagged stable or experimental)
  casegraphen schema get --id <schema-id> | --file <filename> --format json [--output <path>]  (exactly one of --id or --file selects the embedded contract to emit)
  casegraphen memory query|conflicts|candidates --store <dir> --case-space-id <id> --input <memory.query.v0.json> --policy <memory.policy.v0.json> --format json [--output <path>]
  casegraphen memory explain|history|sources --store <dir> --case-space-id <id> --target-id <claim-id> --input <memory.query.v0.json> --policy <memory.policy.v0.json> --format json [--output <path>]
  casegraphen memory source <attach|inspect>
  casegraphen memory source attach --source-record <memory.source_record.v0.json> --source-artifact <path> --format json [--output <path>]  (proposal-only; does not persist bytes or mutate accepted state)
  casegraphen memory source inspect --source-record <memory.source_record.v0.json> --source-artifact <path> --format json [--output <path>]
  casegraphen memory check --input <memory.claim.v0.json> --source-record <memory.source_record.v0.json> --source-artifact <path> --policy <memory.policy.v0.json> --format json [--output <path>]
  casegraphen memory propose --store <dir> --case-space-id <id> --input <memory.claim.v0.json> --source-record <memory.source_record.v0.json> --source-artifact <path> --policy <memory.policy.v0.json> --format json [--output <path>]  (binds the proposal to the replayed CaseSpace; emits an unreviewed evidence-cell proposal; never reviews or applies it)
  casegraphen memory index <rebuild|validate>
  casegraphen memory index rebuild --store <dir> --case-space-id <id> --input <memory.query.v0.json> --policy <memory.policy.v0.json> --format json [--output <path>]
  casegraphen memory index validate --store <dir> --case-space-id <id> --input <memory.query.v0.json> --policy <memory.policy.v0.json> --index <memory.index.v0.json> --format json [--output <path>]
  casegraphen github observe --manifest <github.capture_manifest.v0.json> --capture-dir <dir> [--strict] --format json [--output <path>]  (read-only; observations are never accepted facts; no store access)
  casegraphen github refresh --manifest <new-manifest.json> --capture-dir <dir> --previous-manifest <old-manifest.json> --previous-capture-dir <dir> [--previous-observation <github.pr_observation.v0.json>] [--strict] --format json [--output <path>]  (read-only; observations are never accepted facts; no store access)
  casegraphen github project --manifest <github.capture_manifest.v0.json> --capture-dir <dir> [--require-independent-review] [--strict] --format json [--output <path>]  (read-only; observations are never accepted facts; no store access)
  casegraphen-mcp-host --state <control-plane.state.json> --store <case-store-dir> --artifacts <artifact-dir> [--resource-journal <dir>] [--resource-capacities <resource.allocator_configuration.v0.json>] [--resource-retention-policy <resource.allocator_retention_policy.v0.json> --resource-checkpoint-interval <positive-events>] --auth-token-env <environment-variable>  (experimental Graph Engineering and Memory Plane workflows; resource allocator state is host-canonical; checkpoint maintenance preserves archived full replay; runtime/redesign/memory proposals are never auto-accepted; verification lineage and memory projections are read-only)
  casegraphen lift native --store <dir> --input <native.case.space.json> --revision-id <id> --format json [--output <path>]
  casegraphen lift workflow --store <dir> --input <workflow.graph.json> --revision-id <id> --format json [--output <path>]
  casegraphen lift case-graph --store <dir> --input <case.graph.json> --revision-id <id> --format json [--output <path>]
  casegraphen lift github-issues --store <dir> --input <snapshot.json> --revision-id <id> --format json [--output <path>]
  casegraphen space new --store <dir> --case-space-id <id> --space-id <id> --title <text> --revision-id <id> --format json [--output <path>]
  casegraphen space list --store <dir> --format json [--output <path>]
  casegraphen space inspect|history|replay|validate|frontier|evidence|project --store <dir> --case-space-id <id> --format json [--output <path>]
  casegraphen space reason --store <dir> --case-space-id <id> [--since-revision <revision-id>] [--strict] --format json|text [--output <path>]
  casegraphen space rebuild --store <dir> --case-space-id <id> [--adopt-existing-log] --format json [--output <path>]
  casegraphen space topology --store <dir> --case-space-id <id> --format json [--higher-order [--max-dimension <n>] [--min-persistence <n>]] [--output <path>]
  casegraphen space topology diff --left-store <dir> --left-case-space-id <id> --right-store <dir> --right-case-space-id <id> --format json [--higher-order [--max-dimension <n>] [--min-persistence <n>]] [--output <path>]
  casegraphen morphism propose --store <dir> --case-space-id <id> --input <case_morphism.json> --format json [--output <path>]  (ungated proposal-file write; no durable case-state mutation)
  casegraphen morphism check --store <dir> --case-space-id <id> --morphism-id <id> --format json [--output <path>]
  casegraphen morphism apply --store <dir> --case-space-id <id> --morphism-id <id> --base-revision-id <id> --reviewer-id <id> --reason <text> [--gate-profile <name> --gate-profile-file <path>] [--actor-id <id>] [--capability-id <id> ...] [--operation-scope-id <id>] [--audience audit|system] [--source-boundary-id <id>] --format json [--output <path>]
  casegraphen morphism reject --store <dir> --case-space-id <id> --morphism-id <id> --reviewer-id <id> --reason <text> --revision-id <id> [--gate-profile <name> --gate-profile-file <path>] [--actor-id <id>] [--capability-id <id> ...] [--operation-scope-id <id>] [--audience audit|system] [--source-boundary-id <id>] --format json [--output <path>]
  casegraphen plan propose --store <dir> --case-space-id <id> --input <execution.plan.json> --format json [--output <path>]
  casegraphen plan check --store <dir> --case-space-id <id> --plan-id <id> --format json [--output <path>]
  casegraphen plan accept --store <dir> --case-space-id <id> --plan-id <id> --reviewer-id <id> --reason <text> --base-revision-id <id> [--gate-profile <name> --gate-profile-file <path>] [--actor-id <id>] [--capability-id <id> ...] [--operation-scope-id <id>] [--audience audit|system] [--source-boundary-id <id>] --format json [--output <path>]
  casegraphen plan reject --store <dir> --case-space-id <id> --plan-id <id> --reviewer-id <id> --reason <text> --base-revision-id <id> [--gate-profile <name> --gate-profile-file <path>] [--actor-id <id>] [--capability-id <id> ...] [--operation-scope-id <id>] [--audience audit|system] [--source-boundary-id <id>] --format json [--output <path>]
  casegraphen topology-review accept|reject|reopen --store <dir> --case-space-id <id> --target-id <claim-cell-id> --input <execution.topology.json> --policy-manifest <deployment-policy-manifest.json> --reviewer-id <id> --reason <text> --base-revision-id <id> [--gate-profile <name> --gate-profile-file <path>] [--actor-id <id>] [--capability-id <id> ...] [--operation-scope-id <id>] [--audience audit|system] [--source-boundary-id <id>] --format json [--output <path>]
  casegraphen topology-review inspect --store <dir> --case-space-id <id> --target-id <claim-cell-id> --format json [--output <path>]
  casegraphen binding register --store <dir> --input <worker.binding.json> --format json [--output <path>]
  casegraphen run --step --store <dir> --case-space-id <id> --plan-id <id> --base-revision-id <id> --enable-worker shell [--gate-profile <name> --gate-profile-file <path>] [--actor-id <id>] [--capability-id <id> ...] [--operation-scope-id <id>] [--audience audit|system] [--source-boundary-id <id>] [--gate-actor-id <same-actor-id>] [--retry-step <step-id>] [--supersede-trace <trace-id>]... [--strict] --format json|text [--output <path>]
  casegraphen run --frontier --store <dir> --case-space-id <id> --plan-id <id> --base-revision-id <id> --enable-worker shell [--gate-profile <name> --gate-profile-file <path>] [--actor-id <id>] [--capability-id <id> ...] [--operation-scope-id <id>] [--audience audit|system] [--source-boundary-id <id>] [--gate-actor-id <same-actor-id>] [--max-parallel <n>] [--retry-step <step-id>]... [--supersede-trace <trace-id>]... [--strict] --format json|text [--output <path>]
  casegraphen operate --store <dir> --case-space-id <id> --plan-id <id> --base-revision-id <id> --max-rounds <n> --enable-worker shell [--gate-profile <name> --gate-profile-file <path>] [--actor-id <id>] [--capability-id <id> ...] [--operation-scope-id <id>] [--audience audit|system] [--source-boundary-id <id>] [--gate-actor-id <same-actor-id>] [--max-parallel <n>] [--supersede-trace <trace-id>]... [--strict] --format json|text [--output <path>]  (repeats run --frontier's own round selection until a halt other than progress, bounded by --max-rounds; --retry-step is refused — retry with `run --frontier --retry-step`, then operate again; --max-rounds bounds rounds, not work — a round dispatches up to --max-parallel steps concurrently, so the actual spawn bound for one invocation is max-rounds * max-parallel, reported as result.steps_dispatched)
  casegraphen review accept|reject|reopen|waive --store <dir> --case-space-id <id> --target-id <id> --reviewer-id <id> --reason <text> --base-revision-id <id> [--evidence-id <id>]... [--gate-profile <name> --gate-profile-file <path>] [--actor-id <id>] [--capability-id <id> ...] [--operation-scope-id <id>] [--audience audit|system] [--source-boundary-id <id>] --format json [--output <path>]
  casegraphen evidence attach --store <dir> --case-space-id <id> --base-revision-id <id> --input <evidence-cell.json> [--satisfies <target-id>]... [--artifact <path>]... [--input <evidence-cell.json> [--satisfies <target-id>]... [--artifact <path>]...]... [--gate-profile <name> --gate-profile-file <path>] [--actor-id <id>] [--capability-id <id> ...] [--operation-scope-id <id>] [--audience audit|system] [--source-boundary-id <id>] --format json [--output <path>]
  casegraphen cell transition --store <dir> --case-space-id <id> --base-revision-id <id> --cell-id <id> --to <lifecycle> [--gate-profile <name> --gate-profile-file <path>] [--actor-id <id>] [--capability-id <id> ...] [--operation-scope-id <id>] [--audience audit|system] [--source-boundary-id <id>] [--reason <text>] --format json [--output <path>]
  casegraphen packet apply --store <dir> --case-space-id <id> --base-revision-id <id> --packet <evidence.packet.json> [--gate-profile <name> --gate-profile-file <path>] [--actor-id <id>] [--capability-id <id> ...] [--operation-scope-id <id>] [--audience audit|system] [--source-boundary-id <id>] --format json|text [--output <path>]  (attaches the packet's claim and artifacts, then always pauses for an independent review)
  casegraphen packet resume --store <dir> --case-space-id <id> --base-revision-id <id> --packet <evidence.packet.json> --completed-through <revision-id> [--gate-profile <name> --gate-profile-file <path>] [--actor-id <id>] [--capability-id <id> ...] [--operation-scope-id <id>] [--audience audit|system] [--source-boundary-id <id>] --format json|text [--output <path>]  (refuses unless the claim is accepted by an independent review; then transitions the target cell)
  casegraphen obstruction list --store <dir> --case-space-id <id> [--strict] --format json [--output <path>]
  casegraphen completion candidates --store <dir> --case-space-id <id> --format json [--output <path>]
  casegraphen projection apply --store <dir> --case-space-id <id> --projection <projection.json> --format json [--output <path>]
  casegraphen equivalence check --left-store <dir> --left-case-space-id <id> --right-store <dir> --right-case-space-id <id> --format json [--output <path>]
  casegraphen invariant check --store <dir> --case-space-id <id> [--strict] --format json [--output <path>]
  casegraphen invariant close-check --store <dir> --case-space-id <id> --base-revision-id <id> [--strict] --format json [--close-policy-id <id>] [--gate-profile <name> --gate-profile-file <path>] [--actor-id <id>] [--capability-id <id> ...] [--operation-scope-id <id>] [--audience audit|system] [--source-boundary-id <id>] [--validation-evidence-id <id> ...] [--output <path>]

For gated mutations and runs, actor, capability, operation scope, audience, and source boundary
must each be supplied by an explicit flag or the selected profile. Explicit flags win per field.
