usage:
  casegraphen version
  casegraphen --version
  casegraphen lift native --store <dir> --input <native.case.space.json> --revision-id <id> --format json [--output <path>]
  casegraphen lift workflow --store <dir> --input <workflow.graph.json> --revision-id <id> --format json [--output <path>]
  casegraphen lift case-graph --store <dir> --input <case.graph.json> --revision-id <id> --format json [--output <path>]
  casegraphen lift github-issues --store <dir> --input <snapshot.json> --revision-id <id> --format json [--output <path>]
  casegraphen space new --store <dir> --case-space-id <id> --space-id <id> --title <text> --revision-id <id> --format json [--output <path>]
  casegraphen space list --store <dir> --format json [--output <path>]
  casegraphen space inspect|history|replay|validate|frontier|evidence|project --store <dir> --case-space-id <id> --format json [--output <path>]
  casegraphen space reason --store <dir> --case-space-id <id> [--strict] --format json|text [--output <path>]
  casegraphen space rebuild --store <dir> --case-space-id <id> [--adopt-existing-log] --format json [--output <path>]
  casegraphen space topology --store <dir> --case-space-id <id> --format json [--higher-order [--max-dimension <n>] [--min-persistence <n>]] [--output <path>]
  casegraphen space topology diff --left-store <dir> --left-case-space-id <id> --right-store <dir> --right-case-space-id <id> --format json [--higher-order [--max-dimension <n>] [--min-persistence <n>]] [--output <path>]
  casegraphen morphism propose --store <dir> --case-space-id <id> --input <case_morphism.json> --format json [--output <path>]  (ungated proposal-file write; no durable case-state mutation)
  casegraphen morphism check --store <dir> --case-space-id <id> --morphism-id <id> --format json [--output <path>]
  casegraphen morphism apply --store <dir> --case-space-id <id> --morphism-id <id> --base-revision-id <id> --reviewer-id <id> --reason <text> [--gate-profile <name> --gate-profile-file <path>] [--actor-id <id>] [--capability-id <id> ...] [--operation-scope-id <id>] [--audience audit|system] [--source-boundary-id <id>] --format json [--output <path>]
  casegraphen morphism reject --store <dir> --case-space-id <id> --morphism-id <id> --reviewer-id <id> --reason <text> --revision-id <id> [--gate-profile <name> --gate-profile-file <path>] [--actor-id <id>] [--capability-id <id> ...] [--operation-scope-id <id>] [--audience audit|system] [--source-boundary-id <id>] --format json [--output <path>]
  casegraphen plan propose --store <dir> --case-space-id <id> --input <execution.plan.json> --format json [--output <path>]
  casegraphen plan check --store <dir> --case-space-id <id> --plan-id <id> --format json [--output <path>]
  casegraphen plan accept --store <dir> --case-space-id <id> --plan-id <id> --reviewer-id <id> --reason <text> --base-revision-id <id> [--gate-profile <name> --gate-profile-file <path>] [--actor-id <id>] [--capability-id <id> ...] [--operation-scope-id <id>] [--audience audit|system] [--source-boundary-id <id>] --format json [--output <path>]
  casegraphen plan reject --store <dir> --case-space-id <id> --plan-id <id> --reviewer-id <id> --reason <text> --base-revision-id <id> [--gate-profile <name> --gate-profile-file <path>] [--actor-id <id>] [--capability-id <id> ...] [--operation-scope-id <id>] [--audience audit|system] [--source-boundary-id <id>] --format json [--output <path>]
  casegraphen binding register --store <dir> --input <worker.binding.json> --format json [--output <path>]
  casegraphen run --step --store <dir> --case-space-id <id> --plan-id <id> --base-revision-id <id> --enable-worker shell [--gate-profile <name> --gate-profile-file <path>] [--actor-id <id>] [--capability-id <id> ...] [--operation-scope-id <id>] [--audience audit|system] [--source-boundary-id <id>] [--gate-actor-id <same-actor-id>] [--retry-step <step-id>] [--supersede-trace <trace-id>]... [--strict] --format json [--output <path>]
  casegraphen run --frontier --store <dir> --case-space-id <id> --plan-id <id> --base-revision-id <id> --enable-worker shell [--gate-profile <name> --gate-profile-file <path>] [--actor-id <id>] [--capability-id <id> ...] [--operation-scope-id <id>] [--audience audit|system] [--source-boundary-id <id>] [--gate-actor-id <same-actor-id>] [--max-parallel <n>] [--retry-step <step-id>]... [--supersede-trace <trace-id>]... [--strict] --format json [--output <path>]
  casegraphen operate --store <dir> --case-space-id <id> --plan-id <id> --base-revision-id <id> --max-rounds <n> --enable-worker shell [--gate-profile <name> --gate-profile-file <path>] [--actor-id <id>] [--capability-id <id> ...] [--operation-scope-id <id>] [--audience audit|system] [--source-boundary-id <id>] [--gate-actor-id <same-actor-id>] [--max-parallel <n>] [--supersede-trace <trace-id>]... [--strict] --format json [--output <path>]  (repeats run --frontier's own round selection until a halt other than progress, bounded by --max-rounds; --retry-step is refused — retry with `run --frontier --retry-step`, then operate again; --max-rounds bounds rounds, not work — a round dispatches up to --max-parallel steps concurrently, so the actual spawn bound for one invocation is max-rounds * max-parallel, reported as result.steps_dispatched)
  casegraphen review accept|reject|reopen|waive --store <dir> --case-space-id <id> --target-id <id> --reviewer-id <id> --reason <text> --base-revision-id <id> [--evidence-id <id>]... [--gate-profile <name> --gate-profile-file <path>] [--actor-id <id>] [--capability-id <id> ...] [--operation-scope-id <id>] [--audience audit|system] [--source-boundary-id <id>] --format json [--output <path>]
  casegraphen evidence attach --store <dir> --case-space-id <id> --base-revision-id <id> --input <evidence-cell.json> [--satisfies <target-id>]... [--artifact <path>]... [--input <evidence-cell.json> [--satisfies <target-id>]... [--artifact <path>]...]... [--gate-profile <name> --gate-profile-file <path>] [--actor-id <id>] [--capability-id <id> ...] [--operation-scope-id <id>] [--audience audit|system] [--source-boundary-id <id>] --format json [--output <path>]
  casegraphen cell transition --store <dir> --case-space-id <id> --base-revision-id <id> --cell-id <id> --to <lifecycle> [--gate-profile <name> --gate-profile-file <path>] [--actor-id <id>] [--capability-id <id> ...] [--operation-scope-id <id>] [--audience audit|system] [--source-boundary-id <id>] [--reason <text>] --format json [--output <path>]
  casegraphen packet apply --store <dir> --case-space-id <id> --base-revision-id <id> --packet <evidence.packet.json> [--gate-profile <name> --gate-profile-file <path>] [--actor-id <id>] [--capability-id <id> ...] [--operation-scope-id <id>] [--audience audit|system] [--source-boundary-id <id>] --format json [--output <path>]  (attaches the packet's claim and artifacts, then always pauses for an independent review)
  casegraphen packet resume --store <dir> --case-space-id <id> --base-revision-id <id> --packet <evidence.packet.json> --completed-through <revision-id> [--gate-profile <name> --gate-profile-file <path>] [--actor-id <id>] [--capability-id <id> ...] [--operation-scope-id <id>] [--audience audit|system] [--source-boundary-id <id>] --format json [--output <path>]  (refuses unless the claim is accepted by an independent review; then transitions the target cell)
  casegraphen obstruction list --store <dir> --case-space-id <id> [--strict] --format json [--output <path>]
  casegraphen completion candidates --store <dir> --case-space-id <id> --format json [--output <path>]
  casegraphen projection apply --store <dir> --case-space-id <id> --projection <projection.json> --format json [--output <path>]
  casegraphen equivalence check --left-store <dir> --left-case-space-id <id> --right-store <dir> --right-case-space-id <id> --format json [--output <path>]
  casegraphen invariant check --store <dir> --case-space-id <id> [--strict] --format json [--output <path>]
  casegraphen invariant close-check --store <dir> --case-space-id <id> --base-revision-id <id> [--strict] --format json [--close-policy-id <id>] [--gate-profile <name> --gate-profile-file <path>] [--actor-id <id>] [--capability-id <id> ...] [--operation-scope-id <id>] [--audience audit|system] [--source-boundary-id <id>] [--validation-evidence-id <id> ...] [--output <path>]

For gated mutations and runs, actor, capability, operation scope, audience, and source boundary
must each be supplied by an explicit flag or the selected profile. Explicit flags win per field.
